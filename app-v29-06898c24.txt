function renderInline(s){
 s=esc(s);
 s=s.replace(/`([^`\n]+)`/g,(m,c)=>'<code>'+c+'</code>');
 s=s.replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>');
 s=s.replace(/__([^_]+)__/g,'<strong>$1</strong>');
 s=s.replace(/(^|[^*\w])\*([^*\n]+)\*/g,'$1<em>$2</em>');
 s=s.replace(/\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g,'<a href="$2" target="_blank" rel="noopener">$1</a>');
 s=s.replace(/(https?:\/\/[^\s<]+)/g,m=>{
   if(m.indexOf('">')>-1)return m;
   return '<a href="'+m+'" target="_blank" rel="noopener">'+m+'</a>';
 });
 return s;
}

function renderMarkdown(src){
 let text=String(src||'');
 let blocks=text.split(/```/);
 let html='';
 let inCode=false;

 for(let bi=0;bi<blocks.length;bi++){
   let blk=blocks[bi];

   if(inCode){
     let nl=blk.indexOf('\n');
     let lang='';
     let code=blk;
     if(nl>-1){
       lang=blk.slice(0,nl).trim();
       code=blk.slice(nl+1);
     }
     // unclosed fence at the very end: everything is code
     html+='<div class="codeblock"><div class="codehead"><span>'+esc(lang||'code')+'</span><button type="button" class="copycode">\u0646\u0633\u062e</button></div><pre><code>'+esc(code)+'</code></pre></div>';
     inCode=false;
     continue;
   }

   if(bi<blocks.length-1) inCode=true; // next block is code (fence opened)

   let lines=blk.split('\n');
   let i=0;
   let para=[];

   function flushPara(){
     if(para.length){
       html+='<p>'+renderInline(para.join('\n')).replace(/\n/g,'<br>')+'</p>';
       para=[];
     }
   }

   while(i<lines.length){
     let line=lines[i];
     let t=line.trim();

     if(!t){
       flushPara();
       i++;
       continue;
     }

     let h=t.match(/^(#{1,4})\s+(.*)$/);
     if(h){
       flushPara();
       let lvl=h[1].length;
       html+='<h'+lvl+'>'+renderInline(h[2])+'</h'+lvl+'>';
       i++;
       continue;
     }

     if(/^(-{3,}|\*{3,})$/.test(t)){
       flushPara();
       html+='<hr>';
       i++;
       continue;
     }

     if(/^&gt;\s?/.test(renderInline(t))||/^>\s?/.test(t)){
       flushPara();
       let quote=[];
       while(i<lines.length&&/^>\s?/.test(lines[i].trim())){
         quote.push(lines[i].trim().replace(/^>\s?/,''));
         i++;
       }
       html+='<blockquote>'+renderInline(quote.join('\n')).replace(/\n/g,'<br>')+'</blockquote>';
       continue;
     }

     if(/^[-*\u2022]\s+/.test(t)){
       flushPara();
       let items=[];
       while(i<lines.length&&/^[-*\u2022]\s+/.test(lines[i].trim())){
         items.push(lines[i].trim().replace(/^[-*\u2022]\s+/,''));
         i++;
       }
       html+='<ul>'+items.map(x=>'<li>'+renderInline(x)+'</li>').join('')+'</ul>';
       continue;
     }

     if(/^\d+[.)]\s+/.test(t)){
       flushPara();
       let items=[];
       while(i<lines.length&&/^\d+[.)]\s+/.test(lines[i].trim())){
         items.push(lines[i].trim().replace(/^\d+[.)]\s+/,''));
         i++;
       }
       html+='<ol>'+items.map(x=>'<li>'+renderInline(x)+'</li>').join('')+'</ol>';
       continue;
     }

     para.push(line);
     i++;
   }

   flushPara();
 }

 return html;
}

function revealText(b,full){
 let i=0;
 let step=Math.max(3,Math.floor(full.length/180));
 let cur=document.createElement('span');
 cur.className='cursor';
 b.appendChild(cur);
 clearInterval(typeTimer);
 typeTimer=setInterval(()=>{
   i=Math.min(full.length,i+step);
   b.innerHTML=renderMarkdown(full.slice(0,i));
   bindCodeCopy(b);
   if(i<full.length){
     b.appendChild(cur);
     scrollDown(true);
   }else{
     clearInterval(typeTimer);
     typeTimer=null;
     b.innerHTML=renderMarkdown(full);
     bindCodeCopy(b);
     scrollDown(true);
   }
 },24);
}

function readFile(file){
 return new Promise((resolve,reject)=>{
   if(!ALLOWED_TYPES.includes(file.type)){
     reject(new Error('\u0646\u0648\u0639 \u0627\u0644\u0645\u0644\u0641 \u063a\u064a\u0631 \u0645\u062f\u0639\u0648\u0645: '+file.name));
     return;
   }
   if(file.size>MAX_FILE_SIZE){
     reject(new Error('\u062d\u062c\u0645 \u0627\u0644\u0645\u0644\u0641 \u0643\u0628\u064a\u0631 \u062c\u062f\u064b\u0627: '+file.name+' (\u0627\u0644\u062d\u062f \u0627\u0644\u0623\u0642\u0635\u0649 20 MB)'));
     return;
   }
   let r=new FileReader();
   r.onload=()=>{
     let z=String(r.result||'');
     let i=z.indexOf(',');
     if(i<0){
       reject(new Error('\u062a\u0639\u0630\u0631 \u0642\u0631\u0627\u0621\u0629 \u0627\u0644\u0645\u0644\u0641'));
       return;
     }
     resolve({
       name:file.name,
       mimeType:file.type||'application/octet-stream',
       data:z.slice(i+1),
       preview:file.type.startsWith('image/')?z:null,
       size:file.size
     });
   };
   r.onerror=()=>reject(new Error('\u062a\u0639\u0630\u0631 \u0642\u0631\u0627\u0621\u0629 \u0627\u0644\u0645\u0644\u0641: '+file.name));
   r.readAsDataURL(file);
 });
}

function isImagePrompt(text){
 const t=String(text||'').trim();
 if(!t) return false;
 const action=/(?:\u0623\u0646\u0634\u0626|\u0627\u0646\u0634\u0626|\u0627\u0635\u0646\u0639|\u0625\u0635\u0646\u0639|\u0627\u0639\u0645\u0644|\u0623\u0639\u0645\u0644|\u0635\u0645\u0645|\u0627\u0635\u0645\u0645|\u0635\u0645\u0651\u0645|\u0627\u0631\u0633\u0645|\u0648\u0644\u0651\u062f|\u0648\u0644\u062f|generate|create|design|make|draw|render)/i;
 const image=/(?:\u0635\u0648\u0631\u0629|\u0635\u0648\u0631|\u062a\u0635\u0645\u064a\u0645|\u0628\u0648\u0633\u062a\u0631|\u0645\u0644\u0635\u0642|\u0644\u0648\u062c\u0648|\u0634\u0639\u0627\u0631|\u0625\u0646\u0641\u0648\u062c\u0631\u0627\u0641\u064a\u0643|\u0627\u0646\u0641\u0648\u062c\u0631\u0627\u0641\u064a\u0643|\u062e\u0644\u0641\u064a\u0629|\u063a\u0644\u0627\u0641|\u0628\u0646\u0631|image|images|picture|poster|logo|illustration|infographic|banner|cover)/i;
 return action.test(t) && image.test(t);
}

function saveCurrent(silent){
 try{
   if(!getSettings().saveHistory)return;
   if(!history.length)return;

   let a=getChats();
   let first=history.find(x=>x.role==='user');
   if(!first)return;

   let existing=a.find(x=>x.id===currentChatId);

   if(existing){
     existing.messages=history.slice(0,60);
     existing.title=existing.customTitle||first.text.replace(/\s+/g,' ').slice(0,42);
     existing.date=new Date().toISOString();
     existing.upd=Date.now();
   }else{
     currentChatId=Date.now();
     a.unshift({
       id:currentChatId,
       title:first.text.replace(/\s+/g,' ').slice(0,42),
       messages:history.slice(0,60),
       date:new Date().toISOString(),
       upd:Date.now(),
       pinned:false
     });
   }

   setChats(a);
   if(getToken())markDirty(currentChatId);
   if(!silent)renderHistory();
 }catch(e){}
}

function isRtlText(t){

}
function fStr(x){return x==null?'':String(x).replace(/\s+\n/g,'\n').trim()}

function fixJsonTail(s){
 var st=[],inStr=false,esc=false;
 for(var i=0;i<s.length;i++){
  var c=s[i];
  if(inStr){if(esc)esc=false;else if(c==='\\')esc=true;else if(c==='"')inStr=false;continue}
  if(c==='"')inStr=true;
  else if(c==='{'||c==='[')st.push(c);
  else if(c==='}'||c===']')st.pop();
 }
 if(inStr)s+='"';
 s=s.replace(/,\s*$/,'').replace(/:\s*$/,':null');
 while(st.length){s+=(st.pop()==='{'?'}':']')}
 return s;
}
function parseFileJson(text){
 var t=String(text||'').replace(/^\uFEFF/,'').trim();
 t=t.replace(/^```[a-zA-Z]*\s*/,'').replace(/```\s*$/,'').trim();
 var a=t.indexOf('{');
 if(a<0)return null;
 var b=t.lastIndexOf('}');
 var body=b>a?t.slice(a,b+1):t.slice(a);
 try{return JSON.parse(body)}catch(e){}
 try{return JSON.parse(fixJsonTail(t.slice(a)))}catch(e){}
 return null;
}
function normTable(tb){

}

   var st=fArr(b.items).map(function(x){x=x||{};return{title:dStr(x.title,60),text:dStr(x.text,140)}}).filter(function(x){return x.title||x.text}).slice(0,8);

  }else if(ty==='bars'){
   var bi=fArr(b.items).map(function(x){x=x||{};var v=Number(String(x.value==null?'':x.value).replace(/[^0-9.\-]/g,''));return{label:dStr(x.label,40),value:v}}).filter(function(x){return x.label&&isFinite(x.value)&&x.value>0}).slice(0,26);
   if(bi.length)blocks.push({type:'bars',title:dStr(b.title,60),unit:dStr(b.unit,20),items:bi});
