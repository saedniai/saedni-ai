/* ---------- Markdown rendering (safe: escapes HTML first) ---------- */
function renderInline(s){
 s=esc(s);
 s=s.replace(/`([^`\n]+)`/g,(m,c)=>'<code>'+c+'</code>');
 s=s.replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>');
 s=s.replace(/__([^_]+)__/g,'<strong>$1</strong>');
 s=s.replace(/(^|[^*\w])\*([^*\n]+)\*/g,'$1<em>$2</em>');
 s=s.replace(/\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g,'<a href="$2" target="_blank" rel="noopener">$1</a>');
 s=s.replace(/(https?:\/\/[^\s<]+)/g,m=>{
   if(m.indexOf('">')>-1)return m;
   return '<a href="'+m+'" target="_blank" rel="noopener">'+m+'</a>';
 });
 return s;
}

function renderMarkdown(src){
 let text=String(src||'');
 let blocks=text.split(/```/);
 let html='';
 let inCode=false;

 for(let bi=0;bi<blocks.length;bi++){
   let blk=blocks[bi];

   if(inCode){
     let nl=blk.indexOf('\n');
     let lang='';
     let code=blk;
     if(nl>-1){
       lang=blk.slice(0,nl).trim();
       code=blk.slice(nl+1);
     }
     // unclosed fence at the very end: everything is code
     html+='<div class="codeblock"><div class="codehead"><span>'+esc(lang||'code')+'</span><button type="button" class="copycode">\u0646\u0633\u062e</button></div><pre><code>'+esc(code)+'</code></pre></div>';
     inCode=false;
     continue;
   }

   if(bi<blocks.length-1) inCode=true; // next block is code (fence opened)

   let lines=blk.split('\n');
   let i=0;
   let para=[];

   function flushPara(){
     if(para.length){
       html+='<p>'+renderInline(para.join('\n')).replace(/\n/g,'<br>')+'</p>';
       para=[];
     }
   }

   while(i<lines.length){
     let line=lines[i];
     let t=line.trim();

     if(!t){
       flushPara();
       i++;
       continue;
     }

     let h=t.match(/^(#{1,4})\s+(.*)$/);
     if(h){
       flushPara();
       let lvl=h[1].length;
       html+='<h'+lvl+'>'+renderInline(h[2])+'</h'+lvl+'>';
       i++;
       continue;
     }

     if(/^(-{3,}|\*{3,})$/.test(t)){
       flushPara();
       html+='<hr>';
       i++;
       continue;
     }

     if(/^&gt;\s?/.test(renderInline(t))||/^>\s?/.test(t)){
       flushPara();
       let quote=[];
       while(i<lines.length&&/^>\s?/.test(lines[i].trim())){
         quote.push(lines[i].trim().replace(/^>\s?/,''));
         i++;
       }
       html+='<blockquote>'+renderInline(quote.join('\n')).replace(/\n/g,'<br>')+'</blockquote>';
       continue;
     }

     if(/^[-*\u2022]\s+/.test(t)){
       flushPara();
       let items=[];
       while(i<lines.length&&/^[-*\u2022]\s+/.test(lines[i].trim())){
         items.push(lines[i].trim().replace(/^[-*\u2022]\s+/,''));
         i++;
       }
       html+='<ul>'+items.map(x=>'<li>'+renderInline(x)+'</li>').join('')+'</ul>';
       continue;
     }

     if(/^\d+[.)]\s+/.test(t)){
       flushPara();
       let items=[];
       while(i<lines.length&&/^\d+[.)]\s+/.test(lines[i].trim())){
         items.push(lines[i].trim().replace(/^\d+[.)]\s+/,''));
         i++;
       }
       html+='<ol>'+items.map(x=>'<li>'+renderInline(x)+'</li>').join('')+'</ol>';
       continue;
     }

     para.push(line);
     i++;
   }

   flushPara();
 }

 return html;
}

function stripMarkdown(s){
 return String(s||'')
   .replace(/```[\s\S]*?```/g,' ')
   .replace(/[#*>`_]/g,'')
   .replace(/\[([^\]]+)\]\(([^)]+)\)/g,'$1')
   .replace(/\s+/g,' ')
   .trim();
}

function readFile(file){
 return new Promise((resolve,reject)=>{
   if(!ALLOWED_TYPES.includes(file.type)){
     reject(new Error('\u0646\u0648\u0639 \u0627\u0644\u0645\u0644\u0641 \u063a\u064a\u0631 \u0645\u062f\u0639\u0648\u0645: '+file.name));
     return;
   }
   if(file.size>MAX_FILE_SIZE){
     reject(new Error('\u062d\u062c\u0645 \u0627\u0644\u0645\u0644\u0641 \u0643\u0628\u064a\u0631 \u062c\u062f\u064b\u0627: '+file.name+' (\u0627\u0644\u062d\u062f \u0627\u0644\u0623\u0642\u0635\u0649 20 MB)'));
     return;
   }
   let r=new FileReader();
   r.onload=()=>{
     let z=String(r.result||'');
     let i=z.indexOf(',');
     if(i<0){
       reject(new Error('\u062a\u0639\u0630\u0631 \u0642\u0631\u0627\u0621\u0629 \u0627\u0644\u0645\u0644\u0641'));
       return;
     }
     resolve({
       name:file.name,
       mimeType:file.type||'application/octet-stream',
       data:z.slice(i+1),
       preview:file.type.startsWith('image/')?z:null,
       size:file.size
     });
   };
   r.onerror=()=>reject(new Error('\u062a\u0639\u0630\u0631 \u0642\u0631\u0627\u0621\u0629 \u0627\u0644\u0645\u0644\u0641: '+file.name));
   r.readAsDataURL(file);
 });
}

function isImagePrompt(text){
 const t=String(text||'').trim();
 if(!t) return false;
 const action=/(?:\u0623\u0646\u0634\u0626|\u0627\u0646\u0634\u0626|\u0627\u0635\u0646\u0639|\u0625\u0635\u0646\u0639|\u0627\u0639\u0645\u0644|\u0623\u0639\u0645\u0644|\u0635\u0645\u0645|\u0627\u0635\u0645\u0645|\u0635\u0645\u0651\u0645|\u0627\u0631\u0633\u0645|\u0648\u0644\u0651\u062f|\u0648\u0644\u062f|generate|create|design|make|draw|render)/i;
 const image=/(?:\u0635\u0648\u0631\u0629|\u0635\u0648\u0631|\u062a\u0635\u0645\u064a\u0645|\u0628\u0648\u0633\u062a\u0631|\u0645\u0644\u0635\u0642|\u0644\u0648\u062c\u0648|\u0634\u0639\u0627\u0631|\u0625\u0646\u0641\u0648\u062c\u0631\u0627\u0641\u064a\u0643|\u0627\u0646\u0641\u0648\u062c\u0631\u0627\u0641\u064a\u0643|\u062e\u0644\u0641\u064a\u0629|\u063a\u0644\u0627\u0641|\u0628\u0646\u0631|image|images|picture|poster|logo|illustration|infographic|banner|cover)/i;

async function sendMessage(forced){
 if(busy)return;

 let p=(forced??input.value).trim();
 if(!p&&!attachments.length)return;

 let sent=attachments.map(({name,mimeType,data,size})=>({name,mimeType,data,size}));
 let preview=attachments.map(({name,mimeType,preview})=>({name,mimeType,preview}));

 add(p||'\u0645\u0631\u0641\u0642','user',true,preview);

 input.value='';
 attachments=[];
 renderAttachments();
 resize();

 busy=true;
 updateSendState();
 refreshRegenerate();
 typing(isImagePrompt(p));

 abortCtrl=new AbortController();
 const settings=getSettings();

 try{
   // Send only the recent, trimmed context to keep responses fast.
   let old=history.slice(0,-1).slice(-MAX_HISTORY_SEND).map(x=>({
     role:x.role,
     text:String(x.text||'').slice(0,MAX_MSG_CHARS)
   }));

   let payload={
     action:'chat',
     prompt:p,
     message:p,
     history:old,
     attachments:sent,
     model:settings.model||'auto',
     token:getToken()
   };

   let r=await fetch(API_URL,{
     method:'POST',
     headers:{'Content-Type':'text/plain;charset=utf-8'},
     body:JSON.stringify(payload),
     signal:abortCtrl.signal
   });

   let raw=await r.text();
   let data;

   try{
     data=JSON.parse(raw);
   }catch(e){
     throw Error('\u0627\u0633\u062a\u062c\u0627\u0628\u0629 \u063a\u064a\u0631 \u0635\u0627\u0644\u062d\u0629 \u0645\u0646 \u0627\u0644\u062e\u0627\u062f\u0645');
   }

   removeTyping();

   if(data.code==='AUTH'){
     forceLogin(data.error);
     return;
   }

   if(!r.ok||data.ok===false){
     throw Error(data.error||('HTTP '+r.status));
   }

   let answer=data.answer??data.response??data.text??data.result??'';

   if(typeof answer!=='string'){
     answer=JSON.stringify(answer,null,2);
   }

   if(data.type==='image' && data.image && data.image.data){
     add(answer||'\u062a\u0645 \u0625\u0646\u0634\u0627\u0621 \u0627\u0644\u0635\u0648\u0631\u0629 \u0628\u0646\u062c\u0627\u062d.','ai',true,[],data.image,Date.now(),true);
   }else{
     add(answer||'\u0644\u0645 \u064a\u0635\u0644 \u0631\u062f \u0645\u0646 \u0627\u0644\u0646\u0645\u0648\u0630\u062c. \u062d\u0627\u0648\u0644 \u0645\u0631\u0629 \u0623\u062e\u0631\u0649.','ai',true,[],null,Date.now(),true);
   }

   saveCurrent();

 }catch(e){
   removeTyping();
   if(e && e.name==='AbortError'){
     add('\u23f9 \u062a\u0645 \u0625\u064a\u0642\u0627\u0641 \u0627\u0644\u062a\u0648\u0644\u064a\u062f.','ai',false);
   }else{
     add(e.message||'\u062e\u0637\u0623 \u063a\u064a\u0631 \u0645\u0639\u0631\u0648\u0641','ai',false);
   }
 }finally{
   busy=false;
   abortCtrl=null;
   updateSendState();
   refreshRegenerate();
   input.focus();
 }
}

function saveCurrent(silent){
 try{
   if(!getSettings().saveHistory)return;
   if(!history.length)return;

   let a=getChats();
   let first=history.find(x=>x.role==='user');
   if(!first)return;

   let existing=a.find(x=>x.id===currentChatId);

   if(existing){
     existing.messages=history.slice(0,60);
     existing.title=existing.customTitle||first.text.replace(/\s+/g,' ').slice(0,42);
     existing.date=new Date().toISOString();
     existing.upd=Date.now();
   }else{
     currentChatId=Date.now();
     a.unshift({
       id:currentChatId,
       title:first.text.replace(/\s+/g,' ').slice(0,42),
       messages:history.slice(0,60),
       date:new Date().toISOString(),
       upd:Date.now(),
       pinned:false
     });
   }

   setChats(a);
   if(getToken())markDirty(currentChatId);
   if(!silent)renderHistory();
 }catch(e){}
}

async function loadSupport(force){

  if(force)document.getElementById('supportMsgs').innerHTML='<div class="support-hint">\u0633\u062C\u0651\u0644 \u062F\u062E\u0648\u0644\u0643 \u0645\u0646 \u0627\u0644\u0642\u0627\u0626\u0645\u0629 \u0627\u0644\u062C\u0627\u0646\u0628\u064A\u0629 \u0639\u0634\u0627\u0646 \u062A\u0643\u0644\u0645 \u0627\u0644\u062F\u0639\u0645.</div>';

  const d=await api({action:'support_poll',token:getToken()});
  if(!d.ok)return;
  const msgs=d.messages||[];
  const sig=msgs.map(function(m){return m.id}).join(',');

  const box=document.getElementById('supportMsgs');
  box.innerHTML='';

  msgs.forEach(function(m){
   const b=document.createElement('div');

 }catch(e){}
}
