function fmt(iso){
  var ms = Date.parse(String(iso||""));
  if(!ms) return "-";
  try { return new Date(ms).toLocaleString("ar-EG",{day:"numeric",month:"short",hour:"numeric",minute:"2-digit"}); } catch(e){ return String(iso); }
}
function day(iso){ var ms = Date.parse(String(iso||"")); if(!ms) return "-"; try { return new Date(ms).toLocaleDateString("ar-EG",{day:"numeric",month:"short",year:"numeric"}); } catch(e){ return "-"; } }

async function api(payload){
  var r = await fetch(API, { method:"POST", headers:{"Content-Type":"text/plain;charset=utf-8"}, body: JSON.stringify(payload) });
  var raw = await r.text();
  try { return JSON.parse(raw); } catch(e){ throw new Error("\u0631\u062F \u063A\u064A\u0631 \u0635\u0627\u0644\u062D \u0645\u0646 \u0627\u0644\u062E\u0627\u062F\u0645"); }
}

async function call(action, extra){
  var p = Object.assign({action:action, adminToken:getToken()}, extra||{});
  var d = await api(p);
  if(d && d.code === "ADMIN_AUTH"){ logout(); throw new Error(d.error || "\u0633\u062C\u0644 \u062F\u062E\u0648\u0644\u0643 \u062A\u0627\u0646\u064A"); }
  return d;
}

function collectPlan(idx){
  var p = state.plans[idx];
  document.querySelectorAll('[data-i="' + idx + '"]').forEach(function(inp){
    var f = inp.getAttribute("data-f");
    var v = inp.value;
    if(f === "features") p.features = v.split("\n").map(function(s){return s.trim();}).filter(Boolean);
    else if(f === "active") p.active = (v === "true");
    else if(f === "price"){ p.priceText = v; p.price = v; }
    else if(f === "durationDays" || f === "order") p[f] = Number(v) || 0;
    else p[f] = v;
  });
  return p;
}

}

async function refreshThread(){
  if(!state.threadId) return;
  try {
    var d = await call("admin_support_thread", { threadId: state.threadId });
    if(!d.ok) return;
    var box = el("msgs");
    if(!box) return;
    var atBottom = box.scrollHeight - box.scrollTop - box.clientHeight < 60;
    box.innerHTML = (d.messages || []).map(function(m){
      return '<div class="m ' + (m.dir === "out" ? "out" : "in") + '">' + esc(m.text) + '<span class="t">' + esc(fmt(m.at)) + '</span></div>';
    }).join("");
    if(atBottom) box.scrollTop = box.scrollHeight;
  } catch(e) {}
}

function reqChip(st){
  if(st === "pending") return '<span class="chip gold">\u0645\u0633\u062A\u0646\u064A</span>';
  if(st === "approved") return '<span class="chip pro">\u0645\u0642\u0628\u0648\u0644</span>';
  if(st === "rejected") return '<span class="chip banned">\u0645\u0631\u0641\u0648\u0636</span>';
  return '<span class="chip">\u0645\u0646\u062A\u0647\u064A</span>';
}

function reqPhoneLink(p){
  var digits = String(p || "").replace(/[^0-9]/g, "");
  if(!digits) return "";
  var intl = digits.indexOf("0") === 0 ? ("20" + digits.slice(1)) : digits;
  return '<a href="https://wa.me/' + intl + '" target="_blank" rel="noopener">' + esc(digits) + '</a>';
}

async function decideRequest(id, decision){
  var days = 0, note = "";
  if(decision === "approve"){
    var inp = el("days_" + id);
    days = Number(inp ? inp.value : 0) || 0;
    if(days < 1){ alert("\u0627\u0643\u062A\u0628 \u0639\u062F\u062F \u0623\u064A\u0627\u0645 \u0635\u062D\u064A\u062D."); return; }
    if(!confirm("\u062A\u0641\u0639\u064A\u0644 \u0627\u0644\u0628\u0627\u0642\u0629 \u0644\u0645\u062F\u0629 " + days + " \u064A\u0648\u0645\u061F")) return;
  } else if(!confirm("\u0631\u0641\u0636 \u0627\u0644\u0637\u0644\u0628\u061F")) return;
  var fn = el("now_" + id);
  var nEl = el("note_" + id);
  note = nEl ? nEl.value.trim() : "";
  try {
    var d = await call("admin_request_decide", { requestId: id, decision: decision, days: days, note: note, fromNow: !!(fn && fn.checked) });
    if(!d.ok) throw new Error(d.error || "\u062A\u0639\u0630\u0631 \u0627\u0644\u062D\u0641\u0638");
    alert(decision === "approve" ? ("\u062A\u0645 \u0627\u0644\u062A\u0641\u0639\u064A\u0644 \u2705 \u0644\u062D\u062F " + fmt(d.paidUntil)) : "\u062A\u0645 \u0631\u0641\u0636 \u0627\u0644\u0637\u0644\u0628");
    loadRequests();
  } catch(e) { alert(e.message); }
}
